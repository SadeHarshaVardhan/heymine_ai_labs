# Two Sum — find indices of two nums that add to target
def two_sum(nums, target):
    seen = {}
    for i, n in enumerate(nums):
        diff = target - n
        if diff in seen:
            return [seen[diff], i]
        seen[n] = i

Time: O(n)  ·  Space: O(n)
